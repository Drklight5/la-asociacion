# Configuracion del arranque en 1 clic (Windows y macOS).
# Editar solo lo de la derecha del "=". Sin espacios ni comillas.

# --- Muse (solo macOS) -------------------------------------------------------
# Nombre del Muse. Dejar vacio si hay un solo Muse encendido cerca.
# Para verlo:  .venv/bin/muselsl list
MUSE_NAME=

# Flags del puente Bluetooth de macOS (muselsl). --acc --gyro --ppg habilitan
# movimiento y ritmo cardiaco reales. Quitar alguno si tu Muse no lo soporta.
MUSE_STREAM_ARGS=--acc --gyro --ppg

# --- Productor -------------------------------------------------------------
# Argumentos extra para producer/muse_producer.py (opcional).
# Ej:  --calibration 45   |   --skip-calibration   |   --debug
PRODUCER_ARGS=

# --- Programas de la obra (Pure Data / Reaper / BlueMuse) -------------------
# El lanzador abre estos programas junto con el resto. Dejar el PATCH / PROJECT
# vacio para NO abrir ese programa.

# Ruta al ejecutable. Vacio = autodetectar en las ubicaciones tipicas
# (Program Files\Pd*, Purr Data, REAPER (x64), etc.). Solo hace falta ponerlo
# si esta instalado en un lugar raro.
PD_EXE=
REAPER_EXE=

# Que se abre al arrancar (rutas relativas a la raiz del repo):
PD_PATCH=pureDataPatch-v0.7/1-Draft.pd
REAPER_PROJECT=reaperProject.RPP

# ReaScript de autoplay: se corre en Reaper unos segundos despues de abrir el
# proyecto (para que termine de cargar). Vacio = Reaper abre sin autoplay.
AUTOPLAY=deploy/autoplay.lua
AUTOPLAY_DELAY=15

# Windows: arrancar BlueMuse y su streaming solo, via bluemuse://startall.
#   si = automatico (ya no hace falta abrir BlueMuse a mano)
#   no = como antes: el lanzador te recuerda abrirlo y darle Start Streaming
BLUEMUSE_AUTO=si

# Al cerrar la ventana principal, cerrar tambien Pd y Reaper (y parar BlueMuse):
#   si = cierre total en 1 paso  |  no = se quedan abiertos (los cerras vos)
# OJO: "si" hace taskkill de TODO pd.exe / reaper.exe de la maquina y puede
# descartar cambios sin guardar en Reaper. Pensado para la maquina del venue.
CERRAR_TODO=si

# --- Grafica -------------------------------------------------------------
# Abrir el navegador con la grafica automaticamente:  si  /  no
ABRIR_NAVEGADOR=si

# --- Puertos (rara vez hace falta tocarlos) --------------------------------
# El productor manda al bridge (BRIDGE_PORT); el bridge reenvia a Pure Data
# en PD_PORT sin tocar nada del patch. WEB_PORT sirve la grafica.
BRIDGE_PORT=9001
PD_PORT=9000
WEB_PORT=8000
